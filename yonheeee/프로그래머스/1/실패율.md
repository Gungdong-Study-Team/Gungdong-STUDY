## 풀이방법
스테이지별 멈춘사람이 있고, 그것을 기반으로 최종까지 못간 실패율을 구하는 문제</br>
문제 이해히는 갓이 관건이었고, 다른 방법 없이 구현</br>

## 코드
```
import java.util.*;

class Solution {
    public int[] solution(int N, int[] stages) {
        int[] answer = new int[N];
        int people = stages.length;
        
        //스테이지별로 숫자 count 배열
        double[] count = new double[N+1];
        double[] rate = new double[N+1];
        
        // 스테이지별 멈춘 사람 count
        for(int i = 1; i <= N; i++){
            for(int j = 0; j < stages.length; j++){
                if(i == stages[j]){
                    count[i]++;
                }
            }
        }
        
        //실패율
        for(int i = 1; i <= N; i++){
            if(people == 0){
                rate[i] = 0;
            }else{
                rate[i] = count[i] / people;
            }
            people -= count[i];
            //System.out.println(rate[i]);
        }
        
        
        // for(int i = 0; i< count.length; i++){
        //      System.out.println(count[i]);
        // }
        
        //내림차순을 위한 Integer
        Integer[] stage = new Integer[N];
        for(int i = 0; i < N; i++){
            stage[i] = i+1;
        }
        
        //내림차순
        Arrays.sort(stage, (a, b) -> {
 
            return Double.compare(rate[b], rate[a]);
        });

        for(int i = 0; i < N; i++){
            answer[i] = stage[i];
        }

        return answer;
    }
}
```

import java.util.*;

class Solution {
    public int[] solution(int N, int[] stages) {
        int[] answer = new int[N];
        int[] count = new int[N + 2];
        
        // 각 스테이지에 멈춰 있는 사람 수
        for (int stage : stages) {
            count[stage]++;
        }

        double[] rate = new double[N + 1];
        int people = stages.length;

        // 실패율 계산
        for (int i = 1; i <= N; i++) {
            if (people == 0) {
                rate[i] = 0;
            } else {
                rate[i] = (double) count[i] / people;
            }

            people -= count[i];
        }

        // 스테이지 번호
        Integer[] stageNum = new Integer[N];

        for (int i = 0; i < N; i++) {
            stageNum[i] = i + 1;
        }

        // 실패율 내림차순, 같으면 스테이지 번호 오름차순
        Arrays.sort(stageNum, (a, b) -> {
            if (rate[a] == rate[b]) {
                return a - b;
            }

            return Double.compare(rate[b], rate[a]);
        });

        // Integer[] -> int[]
        for (int i = 0; i < N; i++) {
            answer[i] = stageNum[i];
        }

        return answer;
    }
}
